from ase.io import read, write
from ase.build import surface
from ase.visualize import view

bulk_file = "full_relax.json"
hkl = (1, 0, 0)
layers = 4
vacuum = 8.0
repeat_xy = (2, 2, 1)

bulk = read(bulk_file)
bulk.pbc = (True, True, True)

slab = surface(bulk, hkl, layers=layers, vacuum=0.0)
slab = slab.repeat(repeat_xy)
slab.center(vacuum=vacuum, axis=2)
slab.pbc = (True, True, True)

write("restart.json", slab)
view(slab)
